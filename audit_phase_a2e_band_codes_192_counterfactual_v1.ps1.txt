def plain(value):

    if isinstance(value, Enum):
        return value.value

    if isinstance(value, BaseModel):
        return {
            str(key): plain(item)
            for key, item in value.__dict__.items()
            if not str(key).startswith("_")

    if isinstance(value, dict):

    r"\b([A-Z]{1,3}\d{2,4})"

        multi_band_pair.finditer(

        return False

        match.group(1).upper(),
        match.group(2).upper(),

        return False

            None,

        return False

def diff_paths(left, right, prefix=""):

    diffs = []

    if type(left) is not type(right):

        diffs.append(
            prefix or "<root>"
        )

        return diffs

    if isinstance(left, dict):

        keys = sorted(
            set(left)
            | set(right)
        )

        for key in keys:

            current = (
                f"{prefix}.{key}"
                if prefix
                else key
            )

            if key not in left:

                diffs.append(
                    current + " [ADDED]"
                )

                continue

            if key not in right:

                diffs.append(
                    current + " [REMOVED]"
                )

                continue

            diffs.extend(
                diff_paths(
                    left[key],
                    right[key],
                    current,
                )
            )

        return diffs

    if isinstance(left, list):

        if left != right:
            diffs.append(
                prefix or "<root>"
            )

        return diffs

    if left != right:

        diffs.append(
            prefix or "<root>"
        )

    return diffs
