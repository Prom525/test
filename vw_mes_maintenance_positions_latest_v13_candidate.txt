 WITH latest_cycle AS (
         SELECT vw_mes_cycle_analysis_clean_v13_candidate.lijn_code,
            vw_mes_cycle_analysis_clean_v13_candidate.locatie,
            vw_mes_cycle_analysis_clean_v13_candidate.band_norm,
            vw_mes_cycle_analysis_clean_v13_candidate.band_station,
            vw_mes_cycle_analysis_clean_v13_candidate.scraper_role,
            vw_mes_cycle_analysis_clean_v13_candidate.physical_position_label_final,
            vw_mes_cycle_analysis_clean_v13_candidate.physical_position_source,
            vw_mes_cycle_analysis_clean_v13_candidate.scraper_type_norm,
            vw_mes_cycle_analysis_clean_v13_candidate.scraper_family,
            vw_mes_cycle_analysis_clean_v13_candidate.cycle_id,
            vw_mes_cycle_analysis_clean_v13_candidate.cycle_start,
            vw_mes_cycle_analysis_clean_v13_candidate.cycle_end,
            vw_mes_cycle_analysis_clean_v13_candidate.meetpunten,
            vw_mes_cycle_analysis_clean_v13_candidate.start_meshoogte_mm,
            vw_mes_cycle_analysis_clean_v13_candidate.eind_meshoogte_mm,
            vw_mes_cycle_analysis_clean_v13_candidate.trend_mm_per_dag,
            vw_mes_cycle_analysis_clean_v13_candidate.cycle_trend_quality,
            vw_mes_cycle_analysis_clean_v13_candidate.slijtage_mm_per_dag_simple,
            vw_mes_cycle_analysis_clean_v13_candidate.has_replace_event,
            vw_mes_cycle_analysis_clean_v13_candidate.maintenance_events,
            vw_mes_cycle_analysis_clean_v13_candidate.commentaar,
            vw_mes_cycle_analysis_clean_v13_candidate.geschatte_dagen_tot_3mm,
            vw_mes_cycle_analysis_clean_v13_candidate.geschatte_datum_3mm,
            vw_mes_cycle_analysis_clean_v13_candidate.status_3mm,
            vw_mes_cycle_analysis_clean_v13_candidate.prioriteit,
            row_number() OVER (PARTITION BY vw_mes_cycle_analysis_clean_v13_candidate.lijn_code, vw_mes_cycle_analysis_clean_v13_candidate.band_norm, vw_mes_cycle_analysis_clean_v13_candidate.band_station, vw_mes_cycle_analysis_clean_v13_candidate.scraper_role, vw_mes_cycle_analysis_clean_v13_candidate.physical_position_label_final, vw_mes_cycle_analysis_clean_v13_candidate.scraper_type_norm ORDER BY vw_mes_cycle_analysis_clean_v13_candidate.cycle_end DESC, vw_mes_cycle_analysis_clean_v13_candidate.cycle_id DESC) AS rn
           FROM vw_mes_cycle_analysis_clean_v13_candidate
          WHERE (vw_mes_cycle_analysis_clean_v13_candidate.cycle_end >= (CURRENT_DATE - '2 years'::interval))
        )
 SELECT lijn_code,
    locatie,
    band_norm,
    band_station,
    scraper_role,
    physical_position_label_final,
    physical_position_source,
    scraper_type_norm,
    scraper_family,
    cycle_id,
    cycle_start,
    cycle_end,
    meetpunten,
    start_meshoogte_mm,
    eind_meshoogte_mm,
    trend_mm_per_dag,
    cycle_trend_quality,
    slijtage_mm_per_dag_simple,
    has_replace_event,
    maintenance_events,
    commentaar,
    geschatte_dagen_tot_3mm,
    geschatte_datum_3mm,
    status_3mm,
    prioriteit,
    rn
   FROM latest_cycle
  WHERE (rn = 1);
